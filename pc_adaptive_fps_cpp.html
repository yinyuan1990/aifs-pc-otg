1. 架构概览

PC 端职责：
  • 采集网络指标（到达率、抖动、丢包、RTT、队列）
  • 运行决策算法，计算目标帧率
  • 通过信令通知 iOS 端执行
  • 接收 iOS 确认，维护状态机

1.1 模块划分

模块 职责 文件建议
NetworkMetricsCollector 采集网络指标 network_metrics.h/.cpp
AdaptiveFpsDecider 决策算法核心 adaptive_fps_decider.h/.cpp
FpsSignalSender 信令发送与确认 fps_signal_sender.h/.cpp
AdaptiveFpsController 总控制器（状态机） adaptive_fps_controller.h/.cpp

2. 数据结构定义

新增 头文件 adaptive_fps_types.h

// adaptive_fps_types.h
#pragma once

#include <cstdint>
#include <string>
#include <chrono>

namespace adaptive {

// ==================== 帧率档位 ====================
enum class FpsLevel {
    FPS_15 = 15,
    FPS_30 = 30,
    FPS_45 = 45,
    FPS_60 = 60
};

constexpr int FPS_LEVELS[] = {15, 30, 45, 60};
constexpr int FPS_LEVEL_COUNT = 4;

// ==================== 紧急度 ====================
enum class Urgency {
    Low,        // 升帧建议，响应时限 ≤ 2000ms
    Normal,     // 常规降帧，响应时限 ≤ 500ms
    High,       // 快速恶化，响应时限 ≤ 200ms
    Critical    // 即将断流，响应时限 ≤ 50ms
};

// ==================== 触发原因 ====================
enum class TriggerReason {
    ArrivalLow,      // 到达率不足
    JitterHigh,      // 抖动过高
    LossHigh,        // 丢包过高
    RttHigh,         // RTT过高
    QueueLow,        // 队列不足
    NetworkRecover,  // 网络恢复
    UserRequest      // 用户请求
};

// ==================== 网络指标结构 ====================
struct NetworkMetrics {
    double arrivalRate;      // 到达率 (0.0 ~ 1.0)
    double jitterRatio;      // 抖动比例 (0.0 ~ 1.0+)
    double lossRate;         // 丢包率 (0.0 ~ 1.0)
    int    rttMs;            // RTT 毫秒
    int    queueDepth;       // 队列帧数
    int    arrivalFps;       // 实际到达帧率
    int64_t timestamp;       // 采集时间戳
};

// ==================== 控制指令结构 ====================
struct FpsControlCommand {
    int           targetFps;
    int           targetBitrate;   // bps，可选
    Urgency       urgency;
    TriggerReason reason;
    int64_t       timestamp;
    uint32_t      seq;
    NetworkMetrics metrics;       // 触发时快照
};

// ==================== iOS 确认结构 ====================
struct FpsAckResponse {
    uint32_t    seq;
    bool        applied;
    int         appliedFps;
    int         appliedBitrate;
    int         execTimeMs;
    std::string encoderState;
};

// ==================== 阈值配置 ====================
struct AdaptiveThresholds {
    // 降帧阈值
    double arrivalRateDropThreshold  = 0.85;   // < 85% 触发降帧
    double jitterRatioDropThreshold  = 0.30;   // > 30% 触发降帧
    double lossRateDropThreshold     = 0.05;   // > 5% 触发降帧
    int    rttDropThresholdMs        = 300;    // > 300ms 触发降帧
    int    queueDepthDropThreshold   = 3;      // < 3帧 触发降帧
    
    // 升帧阈值
    double arrivalRateUpThreshold    = 0.90;   // ≥ 90% 允许升帧
    double jitterRatioUpThreshold    = 0.20;   // < 20% 允许升帧
    int    queueDepthUpThreshold     = 5;      // ≥ 5帧 允许升帧
    int    stableTimeMs              = 1000;   // 稳定 1s 才升帧
    
    // EMA 平滑系数
    double jitterEmaAlpha            = 0.3;    // 抖动 EMA 系数
};

} // namespace adaptive

// network_metrics.h
#pragma once

#include "adaptive_fps_types.h"
#include <deque>
#include <mutex>
#include <chrono>

namespace adaptive {

class NetworkMetricsCollector {
public:
    NetworkMetricsCollector(int targetFps = 60);
    
    // ===== 帧到达时调用 =====
    void onFrameArrived(int64_t timestampMs);
    
    // ===== RTCP 统计更新 =====
    void updateRtcpStats(int packetsLost, int packetsReceived, int rttMs);
    
    // ===== 队列深度更新 =====
    void updateQueueDepth(int depth);
    
    // ===== 获取当前指标 =====
    NetworkMetrics getMetrics() const;
    
    // ===== 设置目标帧率 =====
    void setTargetFps(int fps);

private:
    void calculateJitter(int64_t intervalMs);
    void calculateArrivalRate();
    
    mutable std::mutex m_mutex;
    
    int m_targetFps;
    double m_expectedIntervalMs;  // 1000.0 / targetFps
    
    // 帧到达时间记录
    std::deque<int64_t> m_frameTimestamps;
    static constexpr size_t MAX_FRAME_HISTORY = 120;
    
    // 抖动计算
    double m_jitterEma = 0.0;
    double m_jitterAlpha = 0.3;
    int64_t m_lastFrameTime = 0;
    
    // RTCP 统计
    int m_packetsLost = 0;
    int m_packetsReceived = 0;
    int m_rttMs = 0;
    
    // 队列深度
    int m_queueDepth = 0;
    
    // 到达率计算
    int m_arrivalFps = 0;
    double m_arrivalRate = 1.0;
};

} // namespace adaptive

// network_metrics.cpp
#include "network_metrics.h"
#include <algorithm>
#include <cmath>

namespace adaptive {

NetworkMetricsCollector::NetworkMetricsCollector(int targetFps)
    : m_targetFps(targetFps)
    , m_expectedIntervalMs(1000.0 / targetFps)
{
}

void NetworkMetricsCollector::onFrameArrived(int64_t timestampMs) {
    std::lock_guard<std::mutex> lock(m_mutex);
    
    // 1. 计算帧间隔抖动
    if (m_lastFrameTime > 0) {
        int64_t interval = timestampMs - m_lastFrameTime;
        calculateJitter(interval);
    }
    m_lastFrameTime = timestampMs;
    
    // 2. 记录帧时间戳
    m_frameTimestamps.push_back(timestampMs);
    if (m_frameTimestamps.size() > MAX_FRAME_HISTORY) {
        m_frameTimestamps.pop_front();
    }
    
    // 3. 计算到达率
    calculateArrivalRate();
}

void NetworkMetricsCollector::calculateJitter(int64_t intervalMs) {
    // 瞬时抖动 = |实际间隔 - 期望间隔|
    double jitter = std::abs((double)intervalMs - m_expectedIntervalMs);
    
    // EMA 平滑：J_ema = α * J + (1-α) * J_ema
    m_jitterEma = m_jitterAlpha * jitter + (1.0 - m_jitterAlpha) * m_jitterEma;
}

void NetworkMetricsCollector::calculateArrivalRate() {
    if (m_frameTimestamps.size() < 2) {
        m_arrivalRate = 1.0;
        m_arrivalFps = m_targetFps;
        return;
    }
    
    // 计算最近 1 秒内的帧数
    int64_t now = m_frameTimestamps.back();
    int64_t oneSecondAgo = now - 1000;
    
    int frameCount = 0;
    for (auto it = m_frameTimestamps.rbegin(); it != m_frameTimestamps.rend(); ++it) {
        if (*it >= oneSecondAgo) {
            frameCount++;
        } else {
            break;
        }
    }
    
    m_arrivalFps = frameCount;
    m_arrivalRate = (double)frameCount / m_targetFps;
}

void NetworkMetricsCollector::updateRtcpStats(int packetsLost, int packetsReceived, int rttMs) {
    std::lock_guard<std::mutex> lock(m_mutex);
    m_packetsLost = packetsLost;
    m_packetsReceived = packetsReceived;
    m_rttMs = rttMs;
}

void NetworkMetricsCollector::updateQueueDepth(int depth) {
    std::lock_guard<std::mutex> lock(m_mutex);
    m_queueDepth = depth;
}

NetworkMetrics NetworkMetricsCollector::getMetrics() const {
    std::lock_guard<std::mutex> lock(m_mutex);
    
    NetworkMetrics metrics;
    metrics.arrivalRate = m_arrivalRate;
    metrics.arrivalFps = m_arrivalFps;
    metrics.jitterRatio = m_jitterEma / m_expectedIntervalMs;  // 抖动比例
    
    // 丢包率
    int total = m_packetsLost + m_packetsReceived;
    metrics.lossRate = (total > 0) ? (double)m_packetsLost / total : 0.0;
    
    metrics.rttMs = m_rttMs;
    metrics.queueDepth = m_queueDepth;
    metrics.timestamp = std::chrono::duration_cast<std::chrono::milliseconds>(
        std::chrono::system_clock::now().time_since_epoch()).count();
    
    return metrics;
}

void NetworkMetricsCollector::setTargetFps(int fps) {
    std::lock_guard<std::mutex> lock(m_mutex);
    m_targetFps = fps;
    m_expectedIntervalMs = 1000.0 / fps;
}

} // namespace adaptive

4. 决策算法模块

核心 新增 adaptive_fps_decider.h/.cpp

// adaptive_fps_decider.h
#pragma once

#include "adaptive_fps_types.h"
#include <optional>

namespace adaptive {

class AdaptiveFpsDecider {
public:
    AdaptiveFpsDecider(const AdaptiveThresholds& thresholds = {});
    
    // ===== 核心决策函数 =====
    // 返回 std::nullopt 表示不需要调整
    std::optional<FpsControlCommand> decide(
        const NetworkMetrics& metrics,
        int currentFps,
        int originFps
    );
    
    // ===== 重置状态 =====
    void reset();
    
    // ===== 更新阈值 =====
    void setThresholds(const AdaptiveThresholds& thresholds);

private:
    // 降帧决策
    std::optional<FpsControlCommand> decideDropFps(
        const NetworkMetrics& metrics, int currentFps);
    
    // 升帧决策
    std::optional<FpsControlCommand> decideUpgradeFps(
        const NetworkMetrics& metrics, int currentFps, int originFps);
    
    // 找到最近的档位
    int nearestFpsLevel(int fps);
    
    // 获取下一个更低/更高档位
    int getLowerFpsLevel(int currentFps);
    int getHigherFpsLevel(int currentFps);
    
    // 计算紧急度
    Urgency calculateUrgency(const NetworkMetrics& metrics);
    
    // 计算建议码率
    int calculateBitrate(int fps);
    
    AdaptiveThresholds m_thresholds;
    
    // 状态
    int64_t m_stableStartTime = 0;    // 稳定开始时间
    bool m_isStable = false;           // 是否处于稳定状态
    uint32_t m_seqCounter = 0;         // 序列号计数器
};

} // namespace adaptive

// adaptive_fps_decider.cpp
#include "adaptive_fps_decider.h"
#include <algorithm>
#include <cmath>

namespace adaptive {

AdaptiveFpsDecider::AdaptiveFpsDecider(const AdaptiveThresholds& thresholds)
    : m_thresholds(thresholds)
{
}

std::optional<FpsControlCommand> AdaptiveFpsDecider::decide(
    const NetworkMetrics& metrics,
    int currentFps,
    int originFps)
{
    // ========== 优先检查降帧条件 ==========
    auto dropCmd = decideDropFps(metrics, currentFps);
    if (dropCmd) {
        m_isStable = false;  // 重置稳定状态
        return dropCmd;
    }
    
    // ========== 检查升帧条件 ==========
    auto upCmd = decideUpgradeFps(metrics, currentFps, originFps);
    if (upCmd) {
        return upCmd;
    }
    
    return std::nullopt;  // 不需要调整
}

std::optional<FpsControlCommand> AdaptiveFpsDecider::decideDropFps(
    const NetworkMetrics& metrics, int currentFps)
{
    int targetFps = currentFps;
    TriggerReason reason = TriggerReason::ArrivalLow;
    bool shouldDrop = false;
    
    // ===== 条件1：到达率不足 =====
    if (metrics.arrivalRate < m_thresholds.arrivalRateDropThreshold) {
        // 直接降到到达帧率
        int fpsByArrival = nearestFpsLevel(metrics.arrivalFps);
        if (fpsByArrival < targetFps) {
            targetFps = fpsByArrival;
            reason = TriggerReason::ArrivalLow;
            shouldDrop = true;
        }
    }
    
    // ===== 条件2：抖动过高（30% ~ 100% → 60fps ~ 15fps）=====
    if (metrics.jitterRatio > m_thresholds.jitterRatioDropThreshold) {
        // 线性映射：30% → 60fps, 100% → 15fps
        double ratio = std::min(1.0, metrics.jitterRatio);
        double dropRatio = (ratio - 0.30) / 0.70;  // 0 ~ 1
        int fpsByJitter = (int)(60 - 45 * dropRatio);
        fpsByJitter = nearestFpsLevel(fpsByJitter);
        
        if (fpsByJitter < targetFps) {
            targetFps = fpsByJitter;
            reason = TriggerReason::JitterHigh;
            shouldDrop = true;
        }
    }
    
    // ===== 条件3：丢包过高 =====
    if (metrics.lossRate > m_thresholds.lossRateDropThreshold) {
        int fpsByLoss = (int)(currentFps * 0.7);
        fpsByLoss = nearestFpsLevel(fpsByLoss);
        
        if (fpsByLoss < targetFps) {
            targetFps = fpsByLoss;
            reason = TriggerReason::LossHigh;
            shouldDrop = true;
        }
    }
    
    // ===== 条件4：RTT过高 =====
    if (metrics.rttMs > m_thresholds.rttDropThresholdMs) {
        int fpsByRtt = (int)(currentFps * 0.7);
        fpsByRtt = nearestFpsLevel(fpsByRtt);
        
        if (fpsByRtt < targetFps) {
            targetFps = fpsByRtt;
            reason = TriggerReason::RttHigh;
            shouldDrop = true;
        }
    }
    
    // ===== 条件5：队列不足 =====
    if (metrics.queueDepth < m_thresholds.queueDepthDropThreshold) {
        int fpsByQueue = getLowerFpsLevel(currentFps);
        
        if (fpsByQueue < targetFps) {
            targetFps = fpsByQueue;
            reason = TriggerReason::QueueLow;
            shouldDrop = true;
        }
    }
    
    // 如果需要降帧且目标帧率低于当前帧率
    if (shouldDrop && targetFps < currentFps) {
        FpsControlCommand cmd;
        cmd.targetFps = targetFps;
        cmd.targetBitrate = calculateBitrate(targetFps);
        cmd.urgency = calculateUrgency(metrics);
        cmd.reason = reason;
        cmd.timestamp = metrics.timestamp;
        cmd.seq = ++m_seqCounter;
        cmd.metrics = metrics;
        
        return cmd;
    }
    
    return std::nullopt;
}

std::optional<FpsControlCommand> AdaptiveFpsDecider::decideUpgradeFps(
    const NetworkMetrics& metrics, int currentFps, int originFps)
{
    // 已经是最高帧率
    if (currentFps >= originFps) {
        return std::nullopt;
    }
    
    // ===== 所有升帧条件必须同时满足 =====
    bool canUpgrade = 
        (metrics.jitterRatio < m_thresholds.jitterRatioUpThreshold) &&   // 抖动稳定
        (metrics.arrivalRate >= m_thresholds.arrivalRateUpThreshold) &&  // 到达率充足
        (metrics.queueDepth >= m_thresholds.queueDepthUpThreshold);      // 队列充足
    
    if (!canUpgrade) {
        m_isStable = false;
        return std::nullopt;
    }
    
    // ===== 检查稳定持续时间 =====
    if (!m_isStable) {
        m_isStable = true;
        m_stableStartTime = metrics.timestamp;
        return std::nullopt;  // 刚开始稳定，等待
    }
    
    int64_t stableDuration = metrics.timestamp - m_stableStartTime;
    if (stableDuration < m_thresholds.stableTimeMs) {
        return std::nullopt;  // 稳定时间不够
    }
    
    // ===== 升一档 =====
    int targetFps = getHigherFpsLevel(currentFps);
    
    // 不能超过到达帧率和原始帧率
    targetFps = std::min(targetFps, metrics.arrivalFps);
    targetFps = std::min(targetFps, originFps);
    targetFps = nearestFpsLevel(targetFps);
    
    if (targetFps <= currentFps) {
        return std::nullopt;  // 无法升帧
    }
    
    // 重置稳定计时（下次升帧需要重新累积）
    m_isStable = false;
    
    FpsControlCommand cmd;
    cmd.targetFps = targetFps;
    cmd.targetBitrate = calculateBitrate(targetFps);
    cmd.urgency = Urgency::Low;  // 升帧用低优先级
    cmd.reason = TriggerReason::NetworkRecover;
    cmd.timestamp = metrics.timestamp;
    cmd.seq = ++m_seqCounter;
    cmd.metrics = metrics;
    
    return cmd;
}

int AdaptiveFpsDecider::nearestFpsLevel(int fps) {
    int nearest = FPS_LEVELS[0];
    int minDiff = std::abs(fps - nearest);
    
    for (int i = 1; i < FPS_LEVEL_COUNT; i++) {
        int diff = std::abs(fps - FPS_LEVELS[i]);
        if (diff < minDiff) {
            minDiff = diff;
            nearest = FPS_LEVELS[i];
        }
    }
    return nearest;
}

int AdaptiveFpsDecider::getLowerFpsLevel(int currentFps) {
    for (int i = FPS_LEVEL_COUNT - 1; i >= 0; i--) {
        if (FPS_LEVELS[i] < currentFps) {
            return FPS_LEVELS[i];
        }
    }
    return FPS_LEVELS[0];  // 15fps
}

int AdaptiveFpsDecider::getHigherFpsLevel(int currentFps) {
    for (int i = 0; i < FPS_LEVEL_COUNT; i++) {
        if (FPS_LEVELS[i] > currentFps) {
            return FPS_LEVELS[i];
        }
    }
    return FPS_LEVELS[FPS_LEVEL_COUNT - 1];  // 60fps
}

Urgency AdaptiveFpsDecider::calculateUrgency(const NetworkMetrics& metrics) {
    // 即将断流
    if (metrics.queueDepth <= 1 || metrics.jitterRatio > 0.8) {
        return Urgency::Critical;
    }
    // 快速恶化
    if (metrics.queueDepth <= 2 || metrics.jitterRatio > 0.5 || metrics.lossRate > 0.1) {
        return Urgency::High;
    }
    // 正常降帧
    return Urgency::Normal;
}

int AdaptiveFpsDecider::calculateBitrate(int fps) {
    // 码率建议表（bps）
    switch (fps) {
        case 60: return 10000000;  // 10 Mbps
        case 45: return 7000000;   // 7 Mbps
        case 30: return 5000000;   // 5 Mbps
        case 15: return 2500000;   // 2.5 Mbps
        default: return 5000000;
    }
}

void AdaptiveFpsDecider::reset() {
    m_isStable = false;
    m_stableStartTime = 0;
}

void AdaptiveFpsDecider::setThresholds(const AdaptiveThresholds& thresholds) {
    m_thresholds = thresholds;
}

} // namespace adaptive

// fps_signal_sender.h
#pragma once

#include "adaptive_fps_types.h"
#include <functional>
#include <string>
#include <map>
#include <mutex>

namespace adaptive {

// 信令发送回调
using SendCallback = std::function<bool(const std::string& json)>;

// 确认回调
using AckCallback = std::function<void(const FpsAckResponse& ack)>;

class FpsSignalSender {
public:
    FpsSignalSender(SendCallback sendCallback);
    
    // ===== 发送控制指令 =====
    bool sendCommand(const FpsControlCommand& cmd, AckCallback onAck = nullptr);
    
    // ===== 收到 iOS 确认时调用 =====
    void onAckReceived(const std::string& json);
    
    // ===== 超时检查（定时调用）=====
    void checkTimeouts(int64_t nowMs, int timeoutMs = 500);

private:
    std::string toJson(const FpsControlCommand& cmd);
    FpsAckResponse parseAck(const std::string& json);
    std::string urgencyToString(Urgency u);
    std::string reasonToString(TriggerReason r);
    
    SendCallback m_sendCallback;
    
    struct PendingCommand {
        FpsControlCommand cmd;
        AckCallback callback;
        int64_t sendTime;
    };
    
    std::map<uint32_t, PendingCommand> m_pendingCommands;
    std::mutex m_mutex;
};

} // namespace adaptive

// fps_signal_sender.cpp
#include "fps_signal_sender.h"
#include <sstream>
#include <chrono>

// 可以用 nlohmann/json 或 rapidjson，这里用简单字符串拼接

namespace adaptive {

FpsSignalSender::FpsSignalSender(SendCallback sendCallback)
    : m_sendCallback(std::move(sendCallback))
{
}

bool FpsSignalSender::sendCommand(const FpsControlCommand& cmd, AckCallback onAck) {
    std::string json = toJson(cmd);
    
    if (m_sendCallback && m_sendCallback(json)) {
        if (onAck) {
            std::lock_guard<std::mutex> lock(m_mutex);
            m_pendingCommands[cmd.seq] = {
                cmd, 
                onAck, 
                std::chrono::duration_cast<std::chrono::milliseconds>(
                    std::chrono::system_clock::now().time_since_epoch()).count()
            };
        }
        return true;
    }
    return false;
}

void FpsSignalSender::onAckReceived(const std::string& json) {
    FpsAckResponse ack = parseAck(json);
    
    std::lock_guard<std::mutex> lock(m_mutex);
    auto it = m_pendingCommands.find(ack.seq);
    if (it != m_pendingCommands.end()) {
        if (it->second.callback) {
            it->second.callback(ack);
        }
        m_pendingCommands.erase(it);
    }
}

void FpsSignalSender::checkTimeouts(int64_t nowMs, int timeoutMs) {
    std::lock_guard<std::mutex> lock(m_mutex);
    
    for (auto it = m_pendingCommands.begin(); it != m_pendingCommands.end(); ) {
        if (nowMs - it->second.sendTime > timeoutMs) {
            // 超时，回调失败
            if (it->second.callback) {
                FpsAckResponse failAck;
                failAck.seq = it->first;
                failAck.applied = false;
                it->second.callback(failAck);
            }
            it = m_pendingCommands.erase(it);
        } else {
            ++it;
        }
    }
}

        << "\"targetBitrate\":" << cmd.targetBitrate << ","
        << "\"urgency\":\"" << urgencyToString(cmd.urgency) << "\","
        << "\"reason\":\"" << reasonToString(cmd.reason) << "\","

        << "\"queueDepth\":" << cmd.metrics.queueDepth
        << "}"
        << "}"
        << "}";
    return oss.str();
}

    return ack;

std::string FpsSignalSender::urgencyToString(Urgency u) {
    switch (u) {
        case Urgency::Low: return "low";
        case Urgency::Normal: return "normal";
        case Urgency::High: return "high";
        case Urgency::Critical: return "critical";
        default: return "normal";
    }
}

std::string FpsSignalSender::reasonToString(TriggerReason r) {
    switch (r) {
        case TriggerReason::ArrivalLow: return "arrival_low";
        case TriggerReason::JitterHigh: return "jitter_high";
        case TriggerReason::LossHigh: return "loss_high";
        case TriggerReason::RttHigh: return "rtt_high";
        case TriggerReason::QueueLow: return "queue_low";
        case TriggerReason::NetworkRecover: return "network_recover";
        case TriggerReason::UserRequest: return "user_request";
        default: return "unknown";
    }
}

// adaptive_fps_controller.cpp
#include "adaptive_fps_controller.h"
#include <chrono>

namespace adaptive {

AdaptiveFpsController::AdaptiveFpsController(int originFps, SendCallback sendCallback)
    : m_originFps(originFps)
    , m_currentFps(originFps)
    , m_metricsCollector(std::make_unique<NetworkMetricsCollector>(originFps))
    , m_decider(std::make_unique<AdaptiveFpsDecider>())
    , m_signalSender(std::make_unique<FpsSignalSender>(std::move(sendCallback)))
{
}

AdaptiveFpsController::~AdaptiveFpsController() {
    stop();
}

void AdaptiveFpsController::start() {
    if (m_running) return;
    m_running = true;
    m_loopThread = std::thread(&AdaptiveFpsController::runLoop, this);
}

void AdaptiveFpsController::stop() {
    m_running = false;
    if (m_loopThread.joinable()) {
        m_loopThread.join();
    }
}

void AdaptiveFpsController::runLoop() {
    while (m_running) {
        // 每秒决策一次
        std::this_thread::sleep_for(std::chrono::seconds(1));
        
        if (!m_running) break;
        
        // 1. 获取当前指标
        NetworkMetrics metrics = m_metricsCollector->getMetrics();
        
        // 2. 决策
        auto cmdOpt = m_decider->decide(metrics, m_currentFps, m_originFps);
        
        // 3. 发送指令
        if (cmdOpt) {
            m_signalSender->sendCommand(*cmdOpt, [this](const FpsAckResponse& ack) {
                onAckCallback(ack);
            });
        }
        
        // 4. 超时检查
        int64_t nowMs = std::chrono::duration_cast<std::chrono::milliseconds>(
            std::chrono::system_clock::now().time_since_epoch()).count();
        m_signalSender->checkTimeouts(nowMs);
    }
}

void AdaptiveFpsController::onFrameArrived(int64_t timestampMs) {
    m_metricsCollector->onFrameArrived(timestampMs);
}

void AdaptiveFpsController::updateRtcpStats(int packetsLost, int packetsReceived, int rttMs) {
    m_metricsCollector->updateRtcpStats(packetsLost, packetsReceived, rttMs);
}

void AdaptiveFpsController::updateQueueDepth(int depth) {
    m_metricsCollector->updateQueueDepth(depth);
}

void AdaptiveFpsController::onAckReceived(const std::string& json) {
    m_signalSender->onAckReceived(json);
}

void AdaptiveFpsController::onAckCallback(const FpsAckResponse& ack) {
    if (ack.applied) {
        m_currentFps = ack.appliedFps;
        m_metricsCollector->setTargetFps(ack.appliedFps);
        // 可以记录日志：成功切换到 ack.appliedFps
    } else {
        // 可以记录日志：iOS 拒绝或超时
    }
}

} // namespace adaptive

8. 接入点汇总

接入点 触发位置 调用方法 说明
帧到达 解码后 / 渲染前 onFrameArrived(timestampMs) 每帧必调，用于计算到达率和抖动
队列更新 缓冲队列变化时 updateQueueDepth(depth) 每帧或定期调用
RTCP 统计 WebRTC stats 回调 updateRtcpStats(...) 定期（1-5秒）
iOS Ack DataChannel 消息回调 onAckReceived(json) 收到 iOS 确认时
DataChannel 发送 构造函数回调 SendCallback 指令发送通道
