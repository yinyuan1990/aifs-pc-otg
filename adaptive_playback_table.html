🔄 完整逻辑流程（简化版）

// ========== 第一道保险：播放速度控制 ==========
// v8.3: 延迟阈值顺延+70ms（最低150ms）
if (延迟 < 150ms) {
    播放速度 = 50%;          // 紧急减速
} else if (延迟 < 220ms) {
    播放速度 = 50% + (延迟-150)/(220-150) × 50%;  // 线性恢复
} else if (延迟 <= 470ms) {
    播放速度 = 100% + (延迟-220)/(470-220) × 10%;  // 线性加速
} else {
    播放速度 = 110%;         // 最大加速
}

// ========== 第二道保险：降帧（1:1对应）==========
最佳缓冲 = 原始帧率 × 15%;  // 例：60fps → 9帧
缺少帧数 = 最佳缓冲 - 当前队列帧数;

if (缺少帧数 > 0) {
    // ⭐ 1:1降帧：队列每少1帧，帧率降1fps
    目标帧率 = 原始帧率 - 缺少帧数;
    目标帧率 = max(15, 目标帧率);  // 最低15fps
    触发降帧(目标帧率);
}

// ========== 第三道保险：升帧（按当前帧率5%-10%）==========
if (当前队列帧数 >= 最佳缓冲 && 已降帧) {
    // ⭐ 升帧幅度 = 当前推流帧率 × 5%~10%
    升帧幅度 = 当前帧率 × 0.05 ~ 0.10;
    目标帧率 = 当前帧率 + 升帧幅度;
    
    // ⭐ 不超过到达帧率！
    目标帧率 = min(目标帧率, 到达帧率);
    目标帧率 = min(目标帧率, 原始帧率);
    触发升帧(目标帧率);
}
✅ 已确认逻辑：
1. 降帧：队列每减少1帧 → 帧率降低1fps（1:1对应）
2. 升帧：按当前推流帧率的5%-10%回升
3. 升帧上限：不超过到达帧率

🔮 新方案：预测式降帧（提前介入）

核心思路：监测队列下降速度

下降速度 = 上一秒队列 - 当前队列

正常波动：60fps时队列应该在 6-9帧 波动（±1-2帧/秒）
异常信号：队列快速下降（≥3帧/秒）= 网络正在恶化

预测公式：
预测N秒后队列 = 当前队列 - 下降速度 × N
如果预测值 < 安全阈值 → 提前触发降帧

队列下降速度判断表

下降速度 含义 动作 说明
0 或 负值 队列在增加 可以升帧 网络好
1-2帧/秒 正常波动 保持 网络稳定
≥3帧/秒 快速下降 ⚠️ 预警 网络开始恶化
≥5帧/秒 急剧下降 🛑 立即降帧 网络严重恶化

💡 关键：不是等队列降到临界值才反应，而是监测下降趋势，提前介入！
就像汽车ABS：不是等轮胎锁死才介入，而是检测到即将锁死时就提前介入。

int dropSpeed = lastQueueDepth - currentQueueDepth;  // 下降速度

// 预测2秒后队列
int predictQueue = currentQueueDepth - dropSpeed * 2;

// ========== 降帧条件 ==========
bool condition1 = (currentQueueDepth < 3);           // 包量不足
bool condition2 = (appDelayMs > 200);                // 延迟过高
bool condition3 = (dropSpeed >= 3);                  // 下降过快 ⭐新增
bool condition4 = (predictQueue < 0);                // 预测见底 ⭐新增

if (condition1 || condition2 || condition3 || condition4) {
    // 触发降帧
    降帧幅度 = max(包量计算, 延迟计算, 下降速度计算);
}

// ========== 升帧条件 ==========
bool canRestore = (currentQueueDepth >= 5)           // 包量充足
               && (appDelayMs >= 200 && appDelayMs <= 400)  // 延迟正常
               && (dropSpeed <= 0);                  // 队列稳定或增加 ⭐新增

if (canRestore && m_requestedFps > 0) {
    // 触发升帧
    targetFps = currentFps + arrivalFps * 0.10;
}

m_lastQueueDepth = currentQueueDepth;  // 保存供下次计算
