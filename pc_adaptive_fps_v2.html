1. 架构概览

核心特性：
  • 预测式降帧：通过趋势分析，提前 1-2 秒预警
  • 快降慢升：降帧大幅度（四档跳），升帧小幅度（5-10fps）
  • WebSocket 信令：复用现有 SRS 连接

2. 数据结构定义

// adaptive_fps_types.h
#pragma once

#include <cstdint>
#include <string>

namespace adaptive {

// ==================== 帧率四档 ====================
constexpr int FPS_LEVELS[] = {15, 30, 45, 60};
constexpr int FPS_LEVEL_COUNT = 4;

// ==================== 紧急度 ====================
enum class Urgency {
    Critical,   // 紧急：立即执行，≤50ms
    High,       // 高优先级：立即执行，≤200ms
    Normal,     // 正常：可短暂过渡
    Low         // 低优先级：升帧，平滑过渡
};

// ==================== 触发原因 ====================
enum class Reason {
    PredictWarning,    // 预测预警
    PredictCritical,   // 预测紧急
    ArrivalLow,        // 到达率不足
    JitterHigh,        // 抖动过高
    QueueLow,          // 队列不足
    RttHigh,           // RTT过高
    NetworkRecover     // 网络恢复
};

// ==================== 网络指标 ====================
struct NetworkMetrics {
    double arrivalRate;      // 到达率 (0.0 ~ 1.0)
    double jitterRatio;      // 抖动比例
    double jitterEma;        // 抖动 EMA (ms)
    int    arrivalFps;       // 实际到达帧率
    int    queueDepth;       // 队列帧数
    int    rttMs;            // RTT 毫秒
    double lossRate;         // 丢包率
    int64_t timestamp;       // 时间戳
};

// ==================== 趋势指标（新增）====================
struct TrendMetrics {
    double jitterTrend;      // 抖动变化率 (ms/s)
    double arrivalSlope;     // 到达率斜率 (%/s)
    int    queueDropSpeed;   // 队列下降速度 (帧/s)
    double rttTrend;         // RTT 变化率 (ms/s)
    double riskScore;        // 综合风险分 (0~1)
    bool   isPredictDrop;    // 是否预测需要降帧
};

// ==================== set_fps 指令 ====================
struct SetFpsCommand {
    int     fps;
    Urgency urgency;
    Reason  reason;
    int     bitrate;         // bps，0 表示不指定
    int64_t timestamp;
};

// ==================== 阈值配置 ====================
struct AdaptiveConfig {
    // 降帧阈值
    double arrivalRateDropThreshold  = 0.85;
    double jitterRatioDropThreshold  = 0.30;
    int    queueDepthDropThreshold   = 3;
    int    rttDropThresholdMs        = 300;
    
    // 趋势预测阈值
    double jitterTrendThreshold      = 2.0;    // ms/s
    double arrivalSlopeThreshold     = -0.05;  // -5%/s
    int    queueDropSpeedThreshold   = 3;      // 帧/s
    double riskScoreThreshold        = 0.5;
    
    // 升帧阈值
    double arrivalRateUpThreshold    = 0.90;
    double jitterRatioUpThreshold    = 0.20;
    int    queueDepthUpThreshold     = 5;
    int    stableTimeMs              = 1000;
    
    // 升帧幅度
    int    upgradeStep               = 5;      // 每次升 5fps
};

} // namespace adaptive

// network_trend_predictor.h
#pragma once

#include "adaptive_fps_types.h"
#include <deque>
#include <mutex>

namespace adaptive {

class NetworkTrendPredictor {
public:
    NetworkTrendPredictor(const AdaptiveConfig& config = {});
    
    // 更新历史数据（每秒调用一次）
    void update(const NetworkMetrics& metrics);
    
    // 获取趋势分析结果
    TrendMetrics analyze() const;
    
    // 是否需要预警降帧
    bool shouldPreemptiveDrop() const;

private:
    struct Sample {
        int64_t timestamp;
        double jitterEma;
        double arrivalRate;
        int queueDepth;
        int rttMs;
    };
    
    std::deque<Sample> m_history;
    AdaptiveConfig m_config;
    mutable std::mutex m_mutex;
    
    static constexpr size_t HISTORY_SIZE = 5;
};

} // namespace adaptive

// network_trend_predictor.cpp
#include "network_trend_predictor.h"
#include <algorithm>
#include <cmath>

namespace adaptive {

NetworkTrendPredictor::NetworkTrendPredictor(const AdaptiveConfig& config)
    : m_config(config)
{}

void NetworkTrendPredictor::update(const NetworkMetrics& metrics) {
    std::lock_guard<std::mutex> lock(m_mutex);
    
    m_history.push_back({
        metrics.timestamp,
        metrics.jitterEma,
        metrics.arrivalRate,
        metrics.queueDepth,
        metrics.rttMs
    });
    
    while (m_history.size() > HISTORY_SIZE) {
        m_history.pop_front();
    }
}

TrendMetrics NetworkTrendPredictor::analyze() const {
    std::lock_guard<std::mutex> lock(m_mutex);
    
    TrendMetrics trend = {0, 0, 0, 0, 0, false};
    
    if (m_history.size() < 2) {
        return trend;
    }
    
    const auto& current = m_history.back();
    const auto& previous = m_history.front();
    
    double timeDelta = (current.timestamp - previous.timestamp) / 1000.0;
    if (timeDelta <= 0) return trend;
    
    // ===== 计算各项趋势 =====
    trend.jitterTrend = (current.jitterEma - previous.jitterEma) / timeDelta;
    trend.arrivalSlope = (current.arrivalRate - previous.arrivalRate) / timeDelta;
    trend.queueDropSpeed = (int)((previous.queueDepth - current.queueDepth) / timeDelta);
    trend.rttTrend = (current.rttMs - previous.rttMs) / timeDelta;
    
    // ===== 计算风险分 =====
    double jitterScore = std::max(0.0, trend.jitterTrend / 10.0);
    double arrivalScore = std::max(0.0, -trend.arrivalSlope / 0.2);
    double queueScore = std::max(0.0, trend.queueDropSpeed / 6.0);
    double rttScore = std::max(0.0, trend.rttTrend / 200.0);
    
    trend.riskScore = 0.30 * jitterScore + 0.25 * arrivalScore + 
                      0.25 * queueScore + 0.20 * rttScore;
    trend.riskScore = std::min(1.0, trend.riskScore);
    
    // ===== 判断是否预警 =====
    int warningCount = 0;
    if (trend.jitterTrend > m_config.jitterTrendThreshold) warningCount++;
    if (trend.arrivalSlope < m_config.arrivalSlopeThreshold) warningCount++;
    if (trend.queueDropSpeed >= m_config.queueDropSpeedThreshold) warningCount++;
    
    // 满足 2 个条件或风险分 > 阈值
    trend.isPredictDrop = (warningCount >= 2) || (trend.riskScore > m_config.riskScoreThreshold);
    
    return trend;
}

bool NetworkTrendPredictor::shouldPreemptiveDrop() const {
    return analyze().isPredictDrop;
}

} // namespace adaptive

// adaptive_fps_decider.h
#pragma once

#include "adaptive_fps_types.h"
#include <optional>

namespace adaptive {

class AdaptiveFpsDecider {
public:
    AdaptiveFpsDecider(const AdaptiveConfig& config = {});
    
    // 决策：是否需要发送 set_fps 指令
    std::optional<SetFpsCommand> decide(
        const NetworkMetrics& metrics,
        const TrendMetrics& trend,
        int currentFps,
        int originFps
    );
    
    void reset();

private:
    std::optional<SetFpsCommand> decideDropFps(
        const NetworkMetrics& metrics, 
        const TrendMetrics& trend,
        int currentFps);
    
    std::optional<SetFpsCommand> decideUpgradeFps(
        const NetworkMetrics& metrics, 
        const TrendMetrics& trend,
        int currentFps, 
        int originFps);
    
    int getLowerFpsLevel(int fps);
    int calculateBitrate(int fps);
    
    AdaptiveConfig m_config;
    bool m_isStable = false;
    int64_t m_stableStartTime = 0;
};

} // namespace adaptive

// adaptive_fps_decider.cpp
#include "adaptive_fps_decider.h"
#include <algorithm>
#include <chrono>

namespace adaptive {

AdaptiveFpsDecider::AdaptiveFpsDecider(const AdaptiveConfig& config)
    : m_config(config)
{}

std::optional<SetFpsCommand> AdaptiveFpsDecider::decide(
    const NetworkMetrics& metrics,
    const TrendMetrics& trend,
    int currentFps,
    int originFps)
{
    // 优先检查降帧
    auto dropCmd = decideDropFps(metrics, trend, currentFps);
    if (dropCmd) {
        m_isStable = false;
        return dropCmd;
    }
    
    // 检查升帧
    return decideUpgradeFps(metrics, trend, currentFps, originFps);
}

std::optional<SetFpsCommand> AdaptiveFpsDecider::decideDropFps(
    const NetworkMetrics& metrics, 
    const TrendMetrics& trend,
    int currentFps)
{
    int targetFps = currentFps;
    Urgency urgency = Urgency::Normal;
    Reason reason = Reason::ArrivalLow;
    bool shouldDrop = false;
    
    // ========================================
    // ⭐ 条件0：队列即将见底（最高优先级）
    // ========================================
    if (metrics.queueDepth <= 1) {
        targetFps = 15;  // 直接降到最低
        urgency = Urgency::Critical;
        reason = Reason::QueueLow;
        shouldDrop = true;
    }
    // ========================================
    // ⭐ 条件1：预测式降帧（趋势分析）
    // ========================================
    else if (trend.isPredictDrop) {
        if (trend.riskScore > 0.7) {
            // 风险很高，降两档
            targetFps = getLowerFpsLevel(getLowerFpsLevel(currentFps));
            urgency = Urgency::High;
            reason = Reason::PredictCritical;
        } else {
            // 风险中等，降一档
            targetFps = getLowerFpsLevel(currentFps);
            urgency = Urgency::High;
            reason = Reason::PredictWarning;
        }
        shouldDrop = true;
    }
    // ========================================
    // 条件2：到达率不足
    // ========================================
    else if (metrics.arrivalRate < m_config.arrivalRateDropThreshold) {
        targetFps = std::max(15, metrics.arrivalFps);
        urgency = Urgency::Normal;
        reason = Reason::ArrivalLow;
        shouldDrop = true;
    }
    // ========================================
    // 条件3：抖动过高
    // ========================================
    else if (metrics.jitterRatio > m_config.jitterRatioDropThreshold) {
        targetFps = getLowerFpsLevel(currentFps);
        urgency = Urgency::Normal;
        reason = Reason::JitterHigh;
        shouldDrop = true;
    }
    // ========================================
    // 条件4：队列不足
    // ========================================
    else if (metrics.queueDepth < m_config.queueDepthDropThreshold) {
        targetFps = getLowerFpsLevel(currentFps);
        urgency = Urgency::Normal;
        reason = Reason::QueueLow;
        shouldDrop = true;
    }
    // ========================================
    // 条件5：RTT过高
    // ========================================
    else if (metrics.rttMs > m_config.rttDropThresholdMs) {
        targetFps = getLowerFpsLevel(currentFps);
        urgency = Urgency::Normal;
        reason = Reason::RttHigh;
        shouldDrop = true;
    }
    
    // 如果需要降帧且目标帧率低于当前
    if (shouldDrop && targetFps < currentFps) {
        SetFpsCommand cmd;
        cmd.fps = targetFps;
        cmd.urgency = urgency;
        cmd.reason = reason;
        cmd.bitrate = calculateBitrate(targetFps);
        cmd.timestamp = metrics.timestamp;
        return cmd;
    }
    
    return std::nullopt;
}

std::optional<SetFpsCommand> AdaptiveFpsDecider::decideUpgradeFps(
    const NetworkMetrics& metrics, 
    const TrendMetrics& trend,
    int currentFps, 
    int originFps)
{
    // 已经是最高帧率
    if (currentFps >= originFps) {
        return std::nullopt;
    }
    
    // ===== 升帧条件（全部满足）=====
    bool canUpgrade = 
        (metrics.jitterRatio < m_config.jitterRatioUpThreshold) &&
        (trend.jitterTrend <= 0) &&  // 抖动趋势稳定或下降
        (metrics.arrivalRate >= m_config.arrivalRateUpThreshold) &&
        (metrics.queueDepth >= m_config.queueDepthUpThreshold);
    
    if (!canUpgrade) {
        m_isStable = false;
        return std::nullopt;
    }
    
    // 检查稳定持续时间
    if (!m_isStable) {
        m_isStable = true;
        m_stableStartTime = metrics.timestamp;
        return std::nullopt;
    }
    
    if (metrics.timestamp - m_stableStartTime < m_config.stableTimeMs) {
        return std::nullopt;  // 稳定时间不够
    }
    
    // ===== 升帧：每次升 5-10fps =====
    int targetFps = currentFps + m_config.upgradeStep;
    targetFps = std::min(targetFps, metrics.arrivalFps);  // 不超过到达帧率
    targetFps = std::min(targetFps, originFps);           // 不超过原始帧率
    
    if (targetFps <= currentFps) {
        return std::nullopt;
    }
    
    // 重置稳定计时
    m_isStable = false;
    
    SetFpsCommand cmd;
    cmd.fps = targetFps;
    cmd.urgency = Urgency::Low;  // 升帧用低优先级
    cmd.reason = Reason::NetworkRecover;
    cmd.bitrate = calculateBitrate(targetFps);
    cmd.timestamp = metrics.timestamp;
    
    return cmd;
}

int AdaptiveFpsDecider::getLowerFpsLevel(int fps) {
    for (int i = FPS_LEVEL_COUNT - 1; i >= 0; i--) {
        if (FPS_LEVELS[i] < fps) {
            return FPS_LEVELS[i];
        }
    }
    return FPS_LEVELS[0];  // 15fps
}

int AdaptiveFpsDecider::calculateBitrate(int fps) {
    if (fps >= 60) return 10000000;
    if (fps >= 45) return 7000000;
    if (fps >= 30) return 5000000;
    return 2500000;
}

void AdaptiveFpsDecider::reset() {
    m_isStable = false;
    m_stableStartTime = 0;
}

} // namespace adaptive

// adaptive_fps_controller.h
#pragma once

#include "adaptive_fps_types.h"
#include "network_metrics.h"
#include "network_trend_predictor.h"
#include "adaptive_fps_decider.h"
#include <memory>
#include <functional>
#include <thread>
#include <atomic>

namespace adaptive {

// WebSocket 发送回调
using WsSendCallback = std::function<bool(const std::string&)>;

class AdaptiveFpsController {
public:
    AdaptiveFpsController(int originFps, WsSendCallback sendCallback);
    ~AdaptiveFpsController();
    
    void start();
    void stop();
    
    // 外部接口
    void onFrameArrived(int64_t timestampMs);
    void updateQueueDepth(int depth);
    void updateRtcpStats(int packetsLost, int packetsReceived, int rttMs);
    
    int getCurrentFps() const { return m_currentFps; }

private:
    void runLoop();
    std::string buildSetFpsJson(const SetFpsCommand& cmd);
    std::string urgencyToString(Urgency u);
    std::string reasonToString(Reason r);
    
    int m_originFps;
    std::atomic<int> m_currentFps;
    
    std::unique_ptr<NetworkMetricsCollector> m_metrics;
    std::unique_ptr<NetworkTrendPredictor> m_predictor;
    std::unique_ptr<AdaptiveFpsDecider> m_decider;
    
    WsSendCallback m_sendCallback;
    
    std::atomic<bool> m_running{false};
    std::thread m_loopThread;
};

} // namespace adaptive

// adaptive_fps_controller.cpp
#include "adaptive_fps_controller.h"
#include <sstream>
#include <chrono>

namespace adaptive {

AdaptiveFpsController::AdaptiveFpsController(int originFps, WsSendCallback sendCallback)
    : m_originFps(originFps)
    , m_currentFps(originFps)
    , m_metrics(std::make_unique<NetworkMetricsCollector>(originFps))
    , m_predictor(std::make_unique<NetworkTrendPredictor>())
    , m_decider(std::make_unique<AdaptiveFpsDecider>())
    , m_sendCallback(std::move(sendCallback))
{}

AdaptiveFpsController::~AdaptiveFpsController() {
    stop();
}

void AdaptiveFpsController::start() {
    if (m_running) return;
    m_running = true;
    m_loopThread = std::thread(&AdaptiveFpsController::runLoop, this);
}

void AdaptiveFpsController::stop() {
    m_running = false;
    if (m_loopThread.joinable()) {
        m_loopThread.join();
    }
}

void AdaptiveFpsController::runLoop() {
    while (m_running) {
        std::this_thread::sleep_for(std::chrono::seconds(1));
        
        if (!m_running) break;
        
        // 1. 获取当前指标
        NetworkMetrics metrics = m_metrics->getMetrics();
        
        // 2. 更新趋势预测器
        m_predictor->update(metrics);
        TrendMetrics trend = m_predictor->analyze();
        
        // 3. 决策
        auto cmdOpt = m_decider->decide(metrics, trend, m_currentFps, m_originFps);
        
        // 4. 发送指令
        if (cmdOpt && m_sendCallback) {
            std::string json = buildSetFpsJson(*cmdOpt);
            if (m_sendCallback(json)) {
                m_currentFps = cmdOpt->fps;
                m_metrics->setTargetFps(cmdOpt->fps);
            }
        }
    }
}

std::string AdaptiveFpsController::buildSetFpsJson(const SetFpsCommand& cmd) {
    std::ostringstream oss;
    oss << "{"
        << "\"cmd\":\"set_fps\","
        << "\"fps\":" << cmd.fps << ","
        << "\"urgency\":\"" << urgencyToString(cmd.urgency) << "\","
        << "\"reason\":\"" << reasonToString(cmd.reason) << "\","
        << "\"bitrate\":" << cmd.bitrate << ","
        << "\"timestamp\":" << cmd.timestamp
        << "}";
    return oss.str();
}

std::string AdaptiveFpsController::urgencyToString(Urgency u) {
    switch (u) {
        case Urgency::Critical: return "critical";
        case Urgency::High: return "high";
        case Urgency::Normal: return "normal";
        case Urgency::Low: return "low";
        default: return "normal";
    }
}

std::string AdaptiveFpsController::reasonToString(Reason r) {
    switch (r) {
        case Reason::PredictWarning: return "predict_warning";
        case Reason::PredictCritical: return "predict_critical";
        case Reason::ArrivalLow: return "arrival_low";
        case Reason::JitterHigh: return "jitter_high";
        case Reason::QueueLow: return "queue_low";
        case Reason::RttHigh: return "rtt_high";
        case Reason::NetworkRecover: return "network_recover";
        default: return "unknown";
    }
}

}

void AdaptiveFpsController::updateQueueDepth(int depth) {
    m_metrics->updateQueueDepth(depth);
}

void AdaptiveFpsController::updateRtcpStats(int packetsLost, int packetsReceived, int rttMs) {
    m_metrics->updateRtcpStats(packetsLost, packetsReceived, rttMs);
}

8. 总结

PC 端核心逻辑：
  1. 采集指标：帧到达时间、队列深度、RTCP统计
  2. 趋势预测：计算抖动趋势、到达率斜率、队列下降速度
  3. 决策：预测式 + 被动式，取最低帧率
  4. 发送指令：通过 WebSocket 发送 set_fps
降帧策略：快降、大幅度（四档跳）
升帧策略：慢升、小幅度（5fps/次，间隔≥1秒）
